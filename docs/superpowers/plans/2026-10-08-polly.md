# Polly Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers-extended-cc:subagent-driven-development (recommended) or superpowers-extended-cc:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the `fm1-gm` GM sound module into Polly, a QY70-style pocket arranger: 8-track patterns written in C, re-voiced live by a chord track, played in songs with sections, on the FM-1.

**Architecture:** Pure-C, host-tested modules for the musical core (`dsp/chords.c`, `dsp/xpose.c`), an arranger engine (`dsp/arr.c`) ticked inside the audio render in front of the existing Sonivox GM synth (`dsp/gm.c`), and the app (`app/ui.c`, `app/project.c`) on the Felucca platform as in FoMni / Eno / Life. Factory styles are text files compiled to C tables by a Python tool.

**Tech Stack:** C99 (single-precision FPU, no libm, no `double`: the build checks), Felucca's FM-1 platform, Sonivox EAS (Apache-2.0), Python 3 for tools, the host simulator (`host/gm_host.c`, renamed `polly_host.c`) and its scenario scripts, Emscripten for the web emulator.

**User decisions (already made):**
- Name: "Polly".
- "Why not 16 well made?": 16 factory styles, each Main A, Main B, Fill, Ending, our own.
- Factory styles are not taken from Yamaha's ROM or Yamaha's downloadable converted QY70 styles (their licence forbids modification and derivative works); a QY700 importer is "later", for styles you made yourself.
- Drum grooves from Google Magenta's Groove MIDI Dataset (licence to be confirmed: CC BY 4.0 expected), credited.
- Controls: "Let's do the full map": at most three pages per button; sections and templates on SEL held + keys ("a modifier plus black keys could work"); ENV, LFO, ARP, SEQ are pages.
- Sonivox (not tinysynth) for GM: "Let's work towards Sonivox".

**Spec:** `docs/superpowers/specs/2026-10-08-arranger-design.md`. Read it first; this plan does not repeat its reasoning.

## Conventions for every task

- Build and test from the repo root `fm1-gm/` (renamed `fm1-polly/` in Task 0):
  - host tests: `sh tests/run_tests.sh` (prints `all tests passed` at the end, exit 0)
  - firmware: `export PATH=~/github/fm1-parent/bin:$PATH && ./build.sh` (needs `podman machine start` once; ends
    with `package build/polly.fwsc ... identity FM-1_300` and no `FAIL` line)
  - flash (the user allows it): `python3 tools/fm1_install.py build/polly.fwsc --yes` → `done: the FM-1 runs FM-1_300`
- Code style: match the existing files (see `dsp/gm.c`, `app/ui.c`): 4-space indent, comments in plain sentences,
  no `double`, floats as `1.0f`, `fastmath.h` instead of libm. Hardware access only in `firmware/hal/`.
- The audio render is an interrupt on the 8 KiB system stack: no large locals in anything it calls (keep stack
  frames under ~512 bytes; tables `static`).
- Commit after each task with a message ending in the two trailer lines from the session's attribution reminder.

```c
/* SPDX-License-Identifier: GPL-3.0-only */
/* Chords: the QY70's 26 types (manual p.41) with their tones by degree and the scale each implies, names,
 * and recognition from held notes. Pure C, no state: the transposer, the engine and the UI share it. */
#pragma once
#include <stdint.h>

enum {
    CH_MAJ, CH_ADD9, CH_MAJ7, CH_MAJ7_9, CH_6, CH_AUG, CH_7, CH_7_9, CH_7_S9, CH_7_B9, CH_7_13, CH_7_B13,
    CH_7_S11, CH_7_S5, CH_7SUS4, CH_SUS4, CH_MIN, CH_MADD9, CH_MIN7, CH_MIN7_9, CH_MIN7_11, CH_MIN6,
    CH_MINMAJ7, CH_MIN7_B5, CH_DIM, CH_NONE, CH_NTYPES
};
enum { DEG_R, DEG_9, DEG_3, DEG_11, DEG_5, DEG_13, DEG_7, NDEG };   /* in pitch order over the root */
#define CH_NOBASS 0xFF

typedef struct {
    uint8_t root;   /* 0..11, C = 0 */
    uint8_t type;   /* CH_* */
    uint8_t bass;   /* 0..11, or CH_NOBASS */
} chord_t;

const char *chord_type_name(int type);           /* "", "add9", "M7", ... ("---" for CH_NONE) */
char *chord_name(chord_t c, char *buf);          /* "Bm7/E"; buf >= 12 bytes; returns the end */
/* the chord's pitch for each degree, semitones over the root: its own tones where it has them (root, 3rd,
 * 5th, 7th, and the tensions it names), the implied scale's notes for the rest */
const int8_t *chord_degrees(int type);
uint16_t chord_tones(int type);                  /* bit i: i semitones over the root is a chord tone */
/* recognise held notes (MIDI numbers, any order); 1 = a chord in *out, 0 = none (fewer than 3 distinct
 * pitch classes, or no type fits) */
int chord_recognize(const uint8_t *notes, int n, chord_t *out);
```

```c
/* SPDX-License-Identifier: GPL-3.0-only */
#include <stdio.h>
#include <string.h>
#include "chords.h"

static int fails, checks;
#define CHECK(c, ...) do { checks++; if (!(c)) { fails++; printf("FAIL %d: ", __LINE__); printf(__VA_ARGS__); printf("\n"); } } while (0)

static const char *name(int root, int type, int bass)
{
    static char b[16];
    chord_t c = {(uint8_t)root, (uint8_t)type, (uint8_t)bass};
    chord_name(c, b);
    return b;
}

int main(void)
{
    /* names */
    CHECK(!strcmp(name(0, CH_MAJ, CH_NOBASS), "C"), "%s", name(0, CH_MAJ, CH_NOBASS));
    CHECK(!strcmp(name(9, CH_MIN7, CH_NOBASS), "Am7"), "%s", name(9, CH_MIN7, CH_NOBASS));
    CHECK(!strcmp(name(6, CH_7_B9, CH_NOBASS), "F#7(b9)"), "%s", name(6, CH_7_B9, CH_NOBASS));
    CHECK(!strcmp(name(11, CH_MIN7, 4), "Bm7/E"), "%s", name(11, CH_MIN7, 4));
    CHECK(!strcmp(name(7, CH_NONE, CH_NOBASS), "G"), "%s", name(7, CH_NONE, CH_NOBASS));
    CHECK(!strcmp(name(2, CH_MAJ, 2), "D"), "slash equal to the root is not shown: %s", name(2, CH_MAJ, 2));
    CHECK(!strcmp(chord_type_name(CH_MINMAJ7), "mM7"), "mM7");

    /* degree tables: the chord's own tones where it has them */
    {
        const int8_t *d = chord_degrees(CH_MAJ7);
        CHECK(d[DEG_R] == 0 && d[DEG_3] == 4 && d[DEG_5] == 7 && d[DEG_7] == 11, "M7 tones");
        CHECK(d[DEG_9] == 2 && d[DEG_11] == 5 && d[DEG_13] == 9, "M7: ionian between");
        d = chord_degrees(CH_MIN7);
        CHECK(d[DEG_3] == 3 && d[DEG_7] == 10 && d[DEG_9] == 2 && d[DEG_11] == 5 && d[DEG_13] == 9, "m7: dorian");
        d = chord_degrees(CH_7);
        CHECK(d[DEG_3] == 4 && d[DEG_7] == 10 && d[DEG_13] == 9, "7: mixolydian");
        d = chord_degrees(CH_MIN7_B5);
        CHECK(d[DEG_5] == 6 && d[DEG_9] == 1 && d[DEG_13] == 8, "m7(b5): locrian");
        d = chord_degrees(CH_7_B9);
        CHECK(d[DEG_9] == 1, "7(b9) names its 9th");
        d = chord_degrees(CH_SUS4);
        CHECK(d[DEG_3] == 5, "sus4: the 3rd degree is the 4th");
        d = chord_degrees(CH_6);
        CHECK(d[DEG_7] == 9, "6: the 7th degree is the 6th");
        d = chord_degrees(CH_MAJ);
        CHECK(d[DEG_7] == 12, "a triad: the 7th degree folds to the octave");
        d = chord_degrees(CH_DIM);
        CHECK(d[DEG_3] == 3 && d[DEG_5] == 6 && d[DEG_7] == 12, "dim triad");
    }
    CHECK(chord_tones(CH_MAJ7) == (1u << 0 | 1u << 4 | 1u << 7 | 1u << 11), "M7 tone mask");

    /* recognition: every type, every root, root position and every inversion of its core tones */
    for (int t = 0; t < CH_NONE; t++)
        for (int r = 0; r < 12; r++) {
            uint16_t m = chord_tones(t);
            uint8_t core[8];
            int n = 0;
            for (int i = 0; i < 12; i++)
                if (m >> i & 1u)
                    core[n++] = (uint8_t)(i);
            for (int inv = 0; inv < n; inv++) {
                uint8_t notes[8];
                chord_t c;
                for (int i = 0; i < n; i++) {
                    int k = (i + inv) % n, iv = core[k] + (i + inv >= n ? 12 : 0);
                    notes[i] = (uint8_t)(48 + r + iv);
                }
                CHECK(chord_recognize(notes, n, &c) && c.root == r && c.type == t,
                      "type %s root %d inversion %d -> %d %d", chord_type_name(t), r, inv, c.root, c.type);
            }
        }
    /* a slash bass: a note held below the chord, not one of its tones */
    {
        uint8_t n[4] = {40, 59, 62, 66};   /* E2, then B3 D4 F#4: Bm/E */
        chord_t c;
        CHECK(chord_recognize(n, 4, &c) && c.root == 11 && c.type == CH_MIN && c.bass == 4, "Bm/E: %d %d %d", c.root, c.type, c.bass);
    }
    {
        uint8_t n[2] = {60, 64};
        chord_t c;
        CHECK(!chord_recognize(n, 2, &c), "two notes: no chord");
    }
    printf("%d checks, %d failures\n%s\n", checks, fails, fails ? "CHORDS TEST FAILED" : "chords test passed");
    return fails ? 1 : 0;
}
```

```c
/* SPDX-License-Identifier: GPL-3.0-only */
/* Chords (chords.h). */
#include "chords.h"

static const char *const TYPE_NAME[CH_NTYPES] = {"", "add9", "M7", "M7(9)", "6", "aug", "7", "7(9)", "7(#9)", "7(b9)",
    "7(13)", "7(b13)", "7(#11)", "7(#5)", "7sus4", "sus4", "m", "madd9", "m7", "m7(9)", "m7(11)", "m6", "mM7",
    "m7(b5)", "dim", "---"};
static const char *const NOTE[12] = {"C", "C#", "D", "Eb", "E", "F", "F#", "G", "Ab", "A", "Bb", "B"};

/* per type: semitones over the root for R, 9, 3, 11, 5, 13, 7 (DEG_* order). The chord's own tones where it
 * has them; between them, its implied scale: major types ionian (lydian for 7(#11)), 7ths mixolydian,
 * minors dorian (aeolian for m(b13)-less sets), m7(b5) locrian, dim diminished; a triad's missing 7th folds
 * to the octave (12), a 6 chord's 7th degree is its 6th */
static const int8_t DEG[CH_NTYPES][NDEG] = {
    [CH_MAJ] = {0, 2, 4, 5, 7, 9, 12},      [CH_ADD9] = {0, 2, 4, 5, 7, 9, 12},
    [CH_MAJ7] = {0, 2, 4, 5, 7, 9, 11},     [CH_MAJ7_9] = {0, 2, 4, 5, 7, 9, 11},
    [CH_6] = {0, 2, 4, 5, 7, 9, 9},         [CH_AUG] = {0, 2, 4, 6, 8, 9, 12},
    [CH_7] = {0, 2, 4, 5, 7, 9, 10},        [CH_7_9] = {0, 2, 4, 5, 7, 9, 10},
    [CH_7_S9] = {0, 3, 4, 5, 7, 9, 10},     [CH_7_B9] = {0, 1, 4, 5, 7, 9, 10},
    [CH_7_13] = {0, 2, 4, 5, 7, 9, 10},     [CH_7_B13] = {0, 2, 4, 5, 7, 8, 10},
    [CH_7_S11] = {0, 2, 4, 6, 7, 9, 10},    [CH_7_S5] = {0, 2, 4, 5, 8, 9, 10},
    [CH_7SUS4] = {0, 2, 5, 5, 7, 9, 10},    [CH_SUS4] = {0, 2, 5, 5, 7, 9, 12},
    [CH_MIN] = {0, 2, 3, 5, 7, 9, 12},      [CH_MADD9] = {0, 2, 3, 5, 7, 9, 12},
    [CH_MIN7] = {0, 2, 3, 5, 7, 9, 10},     [CH_MIN7_9] = {0, 2, 3, 5, 7, 9, 10},
    [CH_MIN7_11] = {0, 2, 3, 5, 7, 9, 10},  [CH_MIN6] = {0, 2, 3, 5, 7, 9, 9},
    [CH_MINMAJ7] = {0, 2, 3, 5, 7, 9, 11},  [CH_MIN7_B5] = {0, 1, 3, 5, 6, 8, 10},
    [CH_DIM] = {0, 2, 3, 5, 6, 8, 12},      [CH_NONE] = {0, 2, 4, 5, 7, 9, 11},
};
/* the tones that make each chord (bit i: i semitones over the root), for naming and recognition */
static const uint16_t TONES[CH_NTYPES] = {
    [CH_MAJ] = 0x091, [CH_ADD9] = 0x095, [CH_MAJ7] = 0x891, [CH_MAJ7_9] = 0x895, [CH_6] = 0x291,
    [CH_AUG] = 0x111, [CH_7] = 0x491, [CH_7_9] = 0x495, [CH_7_S9] = 0x499, [CH_7_B9] = 0x493,
    [CH_7_13] = 0x691, [CH_7_B13] = 0x591, [CH_7_S11] = 0x4D1, [CH_7_S5] = 0x511, [CH_7SUS4] = 0x4A1,
    [CH_SUS4] = 0x0A1, [CH_MIN] = 0x089, [CH_MADD9] = 0x08D, [CH_MIN7] = 0x489, [CH_MIN7_9] = 0x48D,
    [CH_MIN7_11] = 0x4A9, [CH_MIN6] = 0x289, [CH_MINMAJ7] = 0x889, [CH_MIN7_B5] = 0x449, [CH_DIM] = 0x049,
    [CH_NONE] = 0x001,
};

const char *chord_type_name(int t) { return TYPE_NAME[(unsigned)t < CH_NTYPES ? t : CH_NONE]; }
const int8_t *chord_degrees(int t) { return DEG[(unsigned)t < CH_NTYPES ? t : CH_NONE]; }
uint16_t chord_tones(int t) { return TONES[(unsigned)t < CH_NTYPES ? t : CH_NONE]; }

static char *cat(char *b, const char *s)
{
    while (*s)
        *b++ = *s++;
    *b = 0;
    return b;
}
char *chord_name(chord_t c, char *b)
{
    b = cat(b, NOTE[c.root % 12u]);
    if (c.type != CH_NONE && c.type != CH_MAJ)
        b = cat(b, chord_type_name(c.type));
    if (c.bass != CH_NOBASS && c.bass % 12u != c.root % 12u) {
        b = cat(b, "/");
        b = cat(b, NOTE[c.bass % 12u]);
    }
    return b;
}

static int popcount12(uint32_t m)
{
    int n = 0;
    for (; m; m &= m - 1u)
        n++;
    return n;
}

/* the best fit: each root among the held pitch classes, each type; a type scores for every tone it has
 * that is held, and loses for held notes it does not have and for its own tones missing (its fifth may be
 * missing for free); the simpler type wins a tie (table order). The lowest held note, if it is not a tone
 * of the winner and lies below all the others, is the slash bass, and is left out of the fit. */
int chord_recognize(const uint8_t *notes, int n, chord_t *out)
{
    uint32_t pcs = 0, upper = 0;
    int lo = 128, i, r, t, best = -1000, br = 0, bt = 0, low_pc;
    for (i = 0; i < n; i++)
        if (notes[i] < lo)
            lo = notes[i];
    low_pc = lo % 12;
    for (i = 0; i < n; i++) {
        pcs |= 1u << (notes[i] % 12u);
        if (notes[i] != lo)
            upper |= 1u << (notes[i] % 12u);
    }
    if (popcount12(pcs) < 3)
        return 0;
    for (int pass = 0; pass < 2; pass++) {          /* 0: all notes; 1: the lowest as a slash bass */
        uint32_t set = pass ? upper : pcs;
        if (pass && (popcount12(set) < 3 || (upper >> low_pc & 1u)))
            break;
        for (r = 0; r < 12; r++) {
            if (!(set >> r & 1u))
                continue;
            for (t = 0; t < CH_NONE; t++) {
                uint32_t tm = TONES[t], rel = ((set >> r) | (set << (12 - r))) & 0xFFFu;
                int hit = popcount12(rel & tm), extra = popcount12(rel & ~tm), miss = popcount12(tm & ~rel & ~0x80u);
                int score = hit * 4 - extra * 6 - miss * 5 - (pass ? 3 : 0) - t / 8;
                if (score > best) {
                    best = score;
                    br = r;
                    bt = t;
                    out->bass = pass ? (uint8_t)low_pc : CH_NOBASS;
                }
            }
        }
    }
    if (best < 8)
        return 0;
    out->root = (uint8_t)br;
    out->type = (uint8_t)bt;
    return 1;
}
```

```c
/* SPDX-License-Identifier: GPL-3.0-only */
/* The transposer (xpose.h). The source chord is C M7 over the C major scale: each pitch class is a degree
 * (C R, D 9, E 3, F 11, G 5, A 13, B 7) or a chromatic note just above one (C# = R+1, Eb = 9+1, F# = 11+1,
 * Ab = 5+1, Bb = 13+1). Chord 1 moves each degree to the same degree of the target (chords.c), keeping the
 * chromatic offset and the octave over the root. */
#include "xpose.h"

static const uint8_t SRC_DEG[12] = {DEG_R, DEG_R, DEG_9, DEG_9, DEG_3, DEG_11, DEG_11, DEG_5, DEG_5, DEG_13, DEG_13, DEG_7};
static const uint8_t SRC_SEMI[NDEG] = {0, 2, 4, 5, 7, 9, 11};

void xpose_reset(xstate_t *s)
{
    s->last = -1;
    s->fresh = 1;
}
void xpose_chord_changed(xstate_t *s) { s->fresh = 1; }

static int fold(int n, int lo, int hi)
{
    while (n > hi && n - 12 >= 0)
        n -= 12;
    while (n < lo && n + 12 <= 127)
        n += 12;
    return n;
}
static int nearest(int pc_note, int target)          /* pc_note moved by octaves nearest target */
{
    while (pc_note - target > 6)
        pc_note -= 12;
    while (target - pc_note > 6)
        pc_note += 12;
    return pc_note;
}

static int chord1(const xtrack_t *t, int note, chord_t c)
{
    int rel = note % 12, base = note - rel, d = SRC_DEG[rel], off = rel - SRC_SEMI[d], shift = c.root;
    if ((t->type == XT_CHORD1 || t->type == XT_BASS) && t->hikey < XT_NOKEY && c.root > t->hikey)
        shift -= 12;
    return base + shift + chord_degrees(c.type)[d] + off;
}

int xpose(const xtrack_t *t, xstate_t *s, int note, chord_t c)
{
    int out;
    switch (t->type) {
    case XT_BYPASS: return note;
    case XT_PARALLEL: out = note + c.root; break;
    case XT_CHORD2:
        out = chord1(t, note, c);
        if (s->last >= 0)
            out = nearest(out, s->last);
        break;
    case XT_BASS:
        out = chord1(t, note, c);
        if (s->fresh)
            out = nearest(out - (out % 12) + (c.bass != CH_NOBASS ? c.bass : c.root), out);
        break;
    default:
        out = c.type == CH_NONE ? note + c.root : chord1(t, note, c);
        break;
    }
    out = fold(out, t->lo, t->hi);
    if (out < 0 || out > 127)
        return -1;
    s->last = (int16_t)out;
    s->fresh = 0;
    return out;
}
```
